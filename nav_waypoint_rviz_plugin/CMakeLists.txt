cmake_minimum_required(VERSION 3.8)
project(nav_waypoint_rviz_plugin)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nav_waypoint_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED
  src/add_waypoint_tool.cpp src/waypoint_panel.cpp src/route_execution_panel.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_14)
target_include_directories(${PROJECT_NAME} PUBLIC include)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
ament_target_dependencies(${PROJECT_NAME}
  geometry_msgs tf2 rclcpp rviz_common rviz_default_plugins pluginlib
  nav_waypoint_msgs std_srvs std_msgs tf2_geometry_msgs)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(panel_workflow test/panel_workflow.cpp)
  target_link_libraries(panel_workflow ${PROJECT_NAME} Qt5::Widgets)
  ament_target_dependencies(panel_workflow rclcpp pluginlib rviz_common nav_waypoint_msgs std_srvs)
  set_tests_properties(panel_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ROS_DOMAIN_ID=212;ROS_LOCALHOST_ONLY=1")
  ament_add_gtest(route_execution_workflow test/route_execution_workflow.cpp)
  target_link_libraries(route_execution_workflow ${PROJECT_NAME} Qt5::Widgets)
  ament_target_dependencies(route_execution_workflow rclcpp pluginlib rviz_common nav_waypoint_msgs std_srvs std_msgs)
  set_tests_properties(route_execution_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ROS_DOMAIN_ID=213;ROS_LOCALHOST_ONLY=1")
endif()
ament_package()
